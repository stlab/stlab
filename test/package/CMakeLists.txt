cmake_minimum_required(VERSION 3.24)
project(stlab-consumer LANGUAGES CXX)
enable_testing()
if(NOT DEFINED EXPECT_EXECUTION_SHARED)
  message(FATAL_ERROR "EXPECT_EXECUTION_SHARED must describe the installed execution library.")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../RuntimeDlls.cmake")
list(APPEND CMAKE_PREFIX_PATH "${EXECUTION_PREFIX}")
if(STLAB_PACKAGE_SOURCE)
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  set(CPM_USE_LOCAL_PACKAGES ON CACHE BOOL "" FORCE)
  set(STLAB_INSTALL OFF CACHE BOOL "" FORCE)
  set(STLAB_EXECUTION_INSTALL OFF CACHE BOOL "" FORCE)
  include("${STLAB_PACKAGE_SOURCE}/cmake/CPM.cmake")
  CPMAddPackage(NAME stlab SOURCE_DIR "${STLAB_PACKAGE_SOURCE}" FORCE TRUE)
else()
  find_package(stlab REQUIRED CONFIG)
endif()

get_target_property(execution_imported stlab::execution IMPORTED)
get_target_property(execution_type stlab::execution TYPE)
get_target_property(stlab_imported stlab::stlab IMPORTED)
get_target_property(stlab_type stlab::stlab TYPE)
get_target_property(legacy_type stlab::stlab-core TYPE)
get_target_property(legacy_dependencies stlab::stlab-core INTERFACE_LINK_LIBRARIES)
get_target_property(execution_includes stlab::execution INTERFACE_INCLUDE_DIRECTORIES)
get_target_property(stlab_dependencies stlab::stlab INTERFACE_LINK_LIBRARIES)
if(NOT stlab_dependencies MATCHES "stlab::.*execution" OR
   NOT stlab_dependencies MATCHES "stlab::.*copy-on-write" OR
   NOT stlab_dependencies MATCHES "stlab::.*enum-ops")
  message(FATAL_ERROR "STLab must publicly export all of its package dependencies.")
endif()
if(NOT execution_imported OR TARGET execution OR NOT legacy_type STREQUAL "INTERFACE_LIBRARY")
  message(FATAL_ERROR "Expected one imported execution runtime and an INTERFACE compatibility target.")
endif()
if(EXPECT_EXECUTION_SHARED)
  set(expected_execution_type SHARED_LIBRARY)
else()
  set(expected_execution_type STATIC_LIBRARY)
endif()
if(NOT execution_type STREQUAL expected_execution_type)
  message(FATAL_ERROR "Imported execution is ${execution_type}, expected ${expected_execution_type}.")
endif()
if(STLAB_PACKAGE_SOURCE)
  if(stlab_imported OR NOT TARGET stlab OR NOT TARGET stlab-core)
    message(FATAL_ERROR "Source targets must have the same stlab:: names as installed targets.")
  endif()
  if(DEFINED stlab-execution_SOURCE_DIR AND NOT "${stlab-execution_SOURCE_DIR}" STREQUAL "")
    message(FATAL_ERROR "CPM fetched a second execution source tree.")
  endif()
  if(BUILD_SHARED_LIBS)
    set(expected_stlab_type SHARED_LIBRARY)
  else()
    set(expected_stlab_type STATIC_LIBRARY)
  endif()
  if(NOT stlab_type STREQUAL expected_stlab_type)
    message(FATAL_ERROR "Source STLab is ${stlab_type}, expected ${expected_stlab_type}.")
  endif()
else()
  if(NOT stlab_imported OR TARGET stlab OR TARGET stlab-core)
    message(FATAL_ERROR "Installed STLab targets must be IMPORTED, not source targets.")
  endif()
endif()
if(NOT legacy_dependencies MATCHES "stlab::.*execution" OR legacy_dependencies MATCHES "stlab::stlab")
  message(FATAL_ERROR "Legacy target must refer directly to the same execution dependency.")
endif()
foreach(directory IN LISTS execution_includes)
  string(REGEX REPLACE "^\\$<BUILD_INTERFACE:(.*)>$" "\\1" directory "${directory}")
  if(NOT IS_DIRECTORY "${directory}")
    message(FATAL_ERROR "Execution include directory is not an installed path: ${directory}")
  endif()
  file(REAL_PATH "${directory}" directory)
  file(REAL_PATH "${EXECUTION_PREFIX}" prefix)
  cmake_path(IS_PREFIX prefix "${directory}" NORMALIZE installed)
  if(NOT installed)
    message(FATAL_ERROR "Execution source paths would mask missing installed headers: ${directory}")
  endif()
endforeach()
file(WRITE "${CMAKE_BINARY_DIR}/package-evidence.txt"
  "execution=stlab::execution\nexecution_imported=${execution_imported}\n"
  "execution_type=${execution_type}\nstlab_type=${stlab_type}\n"
  "stlab_imported=${stlab_imported}\nlegacy_type=${legacy_type}\n"
  "legacy_dependencies=${legacy_dependencies}\nstlab_dependencies=${stlab_dependencies}\n"
  "execution_includes=${execution_includes}\n")

foreach(target consumer legacy)
  if(target STREQUAL "consumer")
    add_executable(${target} main.cpp)
    target_link_libraries(${target} PRIVATE stlab::stlab)
  else()
    add_executable(${target} legacy.cpp)
    target_link_libraries(${target} PRIVATE stlab::stlab-core)
  endif()
  target_compile_definitions(${target} PRIVATE
    STLAB_PACKAGE_EXPECT_EXECUTION_SHARED=$<BOOL:${EXPECT_EXECUTION_SHARED}>)
  stlab_copy_runtime_dlls(${target})
  add_test(NAME stlab.package.${target} COMMAND ${target})
  set_tests_properties(stlab.package.${target} PROPERTIES TIMEOUT 10)
endforeach()

if(NOT STLAB_PACKAGE_SOURCE)
  get_target_property(headers stlab::stlab HEADER_SET_headers)
  if(NOT headers)
    message(FATAL_ERROR "The installed STLab header file set is missing.")
  endif()
  file(APPEND "${CMAKE_BINARY_DIR}/package-evidence.txt" "headers=${headers}\n")
  set(header_index 0)
  foreach(header IN LISTS headers)
    if(NOT EXISTS "${header}")
      message(FATAL_ERROR "Missing installed header: ${header}")
    endif()
    string(REGEX REPLACE "^.*[/\\\\]include[/\\\\]" "" canonical "${header}")
    math(EXPR header_index "${header_index} + 1")
    file(WRITE "${CMAKE_BINARY_DIR}/h${header_index}.cpp" "#include <${canonical}>\n")
    add_library(header.${header_index} OBJECT "${CMAKE_BINARY_DIR}/h${header_index}.cpp")
    target_link_libraries(header.${header_index} PRIVATE stlab::stlab)
  endforeach()
endif()
